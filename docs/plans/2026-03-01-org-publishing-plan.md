```typescript
async function getInstallationToken(): Promise<string> {
  const appId = process.env.GITHUB_APP_ID
  const privateKeyPath = process.env.GITHUB_APP_PRIVATE_KEY_PATH
  const installationId = process.env.GITHUB_APP_INSTALLATION_ID

  if (!appId || !privateKeyPath || !installationId) {
    throw new Error('GitHub App env vars not configured: GITHUB_APP_ID, GITHUB_APP_PRIVATE_KEY_PATH, GITHUB_APP_INSTALLATION_ID')
  }

  const privateKey = readFileSync(privateKeyPath, 'utf-8')

  const auth = createAppAuth({
    appId: Number(appId),
    privateKey,
    installationId: Number(installationId),
  })

  const { token } = await auth({ type: 'installation' })
  return token
}
```

**Step 3: Add publishToOrgRepo() function**

This mirrors `publishToPersonalRepo` but uses the org and installation token:

```typescript
async function publishToOrgRepo(
  repoName: string,
  files: Array<{ path: string; content: string }>
): Promise<PublishResult> {
  const token = await getInstallationToken()
  const orgName = process.env.GITHUB_ORG_NAME || 'skill-builder-agent'

  let repoExists = false
  const checkRes = await fetch(`https://api.github.com/repos/${orgName}/${repoName}`, {
    headers: { Authorization: `Bearer ${token}` },
  })
  repoExists = checkRes.ok

  if (!repoExists) {
    const createRes = await fetch(`https://api.github.com/orgs/${orgName}/repos`, {
      method: 'POST',
      headers: {
        Authorization: `Bearer ${token}`,
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({
        name: repoName,
        description: 'Professional identity skill generated by Skill Builder',
        private: false,
        auto_init: true,
      }),
    })
    if (!createRes.ok) {
      const err = await createRes.json().catch(() => ({}))
      console.error('GitHub org repo creation failed:', createRes.status, err)
      throw new Error(`Failed to create org repo: ${createRes.status}`)
    }
    await new Promise((r) => setTimeout(r, 2000))
  }

  for (const file of files) {
    const existingRes = await fetch(
      `https://api.github.com/repos/${orgName}/${repoName}/contents/${file.path}`,
      { headers: { Authorization: `Bearer ${token}` } }
    )
    const existing = existingRes.ok ? await existingRes.json() : null

    const putRes = await fetch(
      `https://api.github.com/repos/${orgName}/${repoName}/contents/${file.path}`,
      {
        method: 'PUT',
        headers: {
          Authorization: `Bearer ${token}`,
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          message: `Update ${file.path}`,
          content: Buffer.from(file.content, 'utf-8').toString('base64'),
          sha: existing?.sha,
        }),
      }
    )
    if (!putRes.ok) {
      const err = await putRes.json().catch(() => ({}))
      console.error(`GitHub file push failed for ${file.path}:`, putRes.status, err)
    }
  }

  return {
    repoUrl: `https://github.com/${orgName}/${repoName}`,
    repoOwner: orgName,
    installCommand: `skills install ${orgName}/${repoName}`,
  }
}
```
